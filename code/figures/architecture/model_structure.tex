% #1 prefix  #2 shift  #3 fill-style: open|state|updated
\newcommand{\river}[3]{%
  \begin{scope}[shift={#2}]
    \coordinate (#1A) at (0.25,2.15);
    \coordinate (#1B) at (1.55,2.45);
    \coordinate (#1C) at (0.95,1.25);
    \coordinate (#1D) at (2.20,1.70);
    \coordinate (#1T) at (1.35,0.35);
    \coordinate (#1E) at (0.15,0.05);
    \coordinate (#1F) at (2.35,0.15);
    \ifnum#3=2
      \foreach \s/\t in {A/C,B/C,C/T,D/T,E/T,F/T}
        \draw[link,line width=1.05pt,-{Latex[length=2.0mm]}] (#1\s) -- (#1\t);
      \draw[downc,line width=0.9pt,-{Latex[length=1.8mm]}]
        ($ (#1T)!0.22!(#1C) $) -- ($ (#1C)!0.18!(#1T) $);
    \else
      \foreach \s/\t in {A/C,B/C,C/T,D/T,E/T,F/T}
        \draw[link,line width=0.9pt,-{Latex[length=1.9mm]}] (#1\s) -- (#1\t);
    \fi
    \foreach \p in {A,B,C,D,E,F} {
      \ifnum#3=0
        \fill[white] (#1\p) circle (0.12);
        \draw[navy,line width=0.7pt] (#1\p) circle (0.12);
      \else
        \fill[gold!80] (#1\p) circle (0.12);
        \draw[gold!50!black,line width=0.6pt] (#1\p) circle (0.12);
      \fi
    }
    \ifnum#3=2
      \fill[teal] (#1T) circle (0.155);
      \draw[teal!50!black,line width=0.6pt] (#1T) circle (0.155);
    \else
      \fill[target] (#1T) circle (0.15);
      \draw[target!60!black,line width=0.6pt] (#1T) circle (0.15);
    \fi
  \end{scope}
}
